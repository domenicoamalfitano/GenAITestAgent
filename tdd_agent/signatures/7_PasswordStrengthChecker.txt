# constructor that assign to the field List<Rules> rules the parameter value.
# Rules is an interface with the method boolean isSatisfiedOn(String string) overrided.
# the class also have the field List<Rule> "BASIC_RULES".
# the basic rules are composed of: ContainsDigits(), ContainsAlphabeticChars(), LongerThan(7), ContainsUpperCaseChars().
PasswordStrengthChecker(List<Rule> rules)

# returns true if the given password satisfies all the rules.
# each rule is checked by invoking isSatisfiedOn(password).
# if at least one rule is not satisfied, the method returns false.
boolean isAcceptable(String password)

# return a PasswordStrengthChecker with the basic rules.
PasswordStrengthChecker basic()

# return a PasswordStrengthChecker with the admin rules, composed of the basic rules, LongerThan(10), ContainsSpecialChars()
PasswordStrengthChecker forAdmin()
